44 auto region = Kokkos::Profiling::ScopedRegion("Extract System Matrices");

46 using ValuesType = typename SystemMatrices<DeviceType>::ValuesType;
47 SystemMatrices<DeviceType> result;
48 const auto num_values = solver.A.values.extent(0);
49
50 // Capture the shared sparsity pattern that is created with the solver.
51 result.row_map = solver.A.graph.row_map;
52 result.col_indices = solver.A.graph.entries;
53
54 // Tangent depends only on state and base parameters — compute once
55 auto params_for_tangent = base_parameters;
56 step::UpdateTangentOperator(params_for_tangent, state);
57
58 // --- Mass pass ---
59 {
60 auto params = base_parameters;
61 params.h = 0.0;
62 params.beta_prime = 1.0;
63 params.gamma_prime = 0.0;
64 params.conditioner = 1.0;
65 params.include_stiffness = false;
66
67 step::ResetSolver(solver);
68 step::UpdateSystemVariables(params, elements, state);
69 step::AssembleSystemMatrix(params, solver, elements);
70
71 result.mass_matrix_values = ValuesType("mass_values", num_values);
72 Kokkos::deep_copy(result.mass_matrix_values, solver.A.values);
73 }
74
75 // --- Stiffness pass ---
76 {
77 auto params = base_parameters;
78 params.h = 0.0;
79 params.beta_prime = 0.0;
80 params.gamma_prime = 0.0;
81 params.conditioner = 1.0;
82 params.include_stiffness = true;
83 params.include_tangent = false;
84

91 }

94 {
95 auto params = base_parameters;
96 params.h = 0.0;

101

103 step::UpdateSystemVariables(params, elements, state);
104 step::AssembleSystemMatrix(params, solver, elements);

108 }
109
110 // --- Constraint pass ---
111 {
112 bool include_tangent{false}; //< post multiply by tangent operator in system matrix
113
114 step::ResetSolver(solver);
115 step::UpdateConstraintVariables(state, constraints, include_tangent);
116 step::AssembleConstraintsMatrix(solver, constraints);
117
118 result.constraint_matrix_values = ValuesType("constraint_values", num_values);
119 Kokkos::deep_copy(result.constraint_matrix_values, solver.A.values);
120 }
121
122 // --- Restore: re-run with original parameters so element
123 // storage is consistent for the next solve step ---
124 {
125 auto params = base_parameters;
126 step::ResetSolver(solver);
127 step::UpdateSystemVariables(params, elements, state);
128
129 // Likely do not need this, but for consistency in having all
130 // updates again in the reset.
131 step::UpdateTangentOperator(params, state);
132
133 step::AssembleSystemMatrix(params, solver, elements);
134 }
135
136 return result;
137}
